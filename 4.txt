#include <stdio.h>

void printPositive(int arr[], int n) {
    for (int i = 0; i < n; i++) {
        if (arr[i] > 0) printf("%d ", arr[i]);
    }
    printf("\n");
}

int main() {
    int n;
    scanf("%d", &n);
    int arr[n];
    for (int i = 0; i < n; i++) scanf("%d", &arr[i]);
    printPositive(arr, n);
    return 0;
}
