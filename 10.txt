#include <stdio.h>

int countAboveAverage(int arr[], int n) {
    if (n <= 0) return 0;
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        sum += arr[i];
    }
    double avg = (double)sum / n;
    int count = 0;
    for (int i = 0; i < n; i++) {
        if ((double)arr[i] > avg) count++;
    }
    return count;
}

int main() {
    int n;
    scanf("%d", &n);
    int arr[n];
    for (int i = 0; i < n; i++) scanf("%d", &arr[i]);
    printf("%d\n", countAboveAverage(arr, n));
    return 0;
}
