#include <stdio.h>

long long sumEven(int arr[], int n) {
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        if (arr[i] % 2 == 0) sum += arr[i];
    }
    return sum;
}

int main() {
    int n;
    scanf("%d", &n);
    int arr[n];
    for (int i = 0; i < n; i++) scanf("%d", &arr[i]);
    printf("%lld\n", sumEven(arr, n));
    return 0;
}
